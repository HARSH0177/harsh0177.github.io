% ── RelieF-CR ─────────────────────────────────────────────────────────────
\textbf{RelieF-CR: Multi-Modal Satellite Cloud Removal} \hfill
{\it PyTorch, Cross-Attention, GANs, Remote Sensing, STN}
\begin{itemize}
  \item Engineered a generative architecture fusing 5.0\,m LISS-IV optical, Sentinel-1 SAR (VV/VH), Sentinel-2 temporal priors, and 4-channel CartoDEM topography via Spatial Transformer Networks (STN) and Windowed Cross-Attention.
  \item Evaluated on a spatial non-overlapping quadrant split across 18,560 patches (2,251 held-out test patches in SE quadrant) to eliminate spatial autocorrelation leakage; achieved \textbf{30.55\,dB PSNR}, \textbf{0.965 SSIM}, and \textbf{1.49$^\circ$ SAM}, outperforming direct concatenation baselines (28.17\,dB).
\end{itemize}
